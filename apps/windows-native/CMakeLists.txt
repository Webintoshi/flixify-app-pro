cmake_minimum_required(VERSION 3.28)
project(FlixifyWindows VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)

find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Quick Qml QuickControls2 Network Concurrent)
find_package(Qt6 6.10 QUIET COMPONENTS Multimedia)
if(BUILD_TESTING)
  find_package(Qt6 6.10 REQUIRED COMPONENTS Test)
endif()
qt_standard_project_setup(REQUIRES 6.10)

# Each source directory contains only production translation units; tests stay separate.
file(GLOB FLIXIFY_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM FLIXIFY_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
file(GLOB FLIXIFY_HEADERS CONFIGURE_DEPENDS src/*.h)
add_library(flixify_desktop STATIC ${FLIXIFY_SOURCES} ${FLIXIFY_HEADERS})
target_include_directories(flixify_desktop PUBLIC src)
target_link_libraries(flixify_desktop PUBLIC Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml
  Qt6::QuickControls2 Qt6::Network Qt6::Concurrent)
target_compile_definitions(flixify_desktop PUBLIC FLIXIFY_APP_VERSION="${PROJECT_VERSION}")
if(TARGET Qt6::Multimedia)
  target_link_libraries(flixify_desktop PUBLIC Qt6::Multimedia)
  target_compile_definitions(flixify_desktop PUBLIC FLIXIFY_HAS_MULTIMEDIA=1)
endif()
if(WIN32)
  target_link_libraries(flixify_desktop PUBLIC crypt32 dwmapi user32)
  target_compile_definitions(flixify_desktop PUBLIC WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
elseif(APPLE)
  target_link_libraries(flixify_desktop PUBLIC "-framework AppKit")
endif()

qt_add_executable(FlixifyWindows src/main.cpp)
target_link_libraries(FlixifyWindows PRIVATE flixify_desktop)
set_target_properties(FlixifyWindows PROPERTIES OUTPUT_NAME "Flixify Pro" WIN32_EXECUTABLE TRUE)

file(GLOB FLIXIFY_QML CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" qml/*.qml)
file(GLOB FLIXIFY_ASSETS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" resources/*.png resources/*.wav)
foreach(asset IN LISTS FLIXIFY_ASSETS)
  get_filename_component(asset_name "${asset}" NAME)
  set_source_files_properties("${asset}" PROPERTIES QT_RESOURCE_ALIAS "brand/${asset_name}")
endforeach()
qt_add_resources(flixify_desktop desktop_resources PREFIX "/" FILES ${FLIXIFY_QML} ${FLIXIFY_ASSETS})

if(WIN32)
  enable_language(RC)
  set(FLIXIFY_WINDOWS_ICON_RC "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico")
  configure_file(resources/windows_app.rc.in windows_app.rc @ONLY)
  target_sources(FlixifyWindows PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows_app.rc")
endif()

foreach(target flixify_desktop FlixifyWindows)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive- /pathmap:${CMAKE_CURRENT_SOURCE_DIR}=C:\\flixify\\source)
    target_link_options(${target} PRIVATE /Brepro /DEBUG:NONE /INCREMENTAL:NO)
  else()
    target_compile_options(${target} PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/flixify/source")
  endif()
endforeach()

if(BUILD_TESTING)
  foreach(name catalog repository player_policy)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE flixify_desktop Qt6::Test)
    add_test(NAME ${name} COMMAND ${name}_test -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/${name}-test.xml,junitxml")
    set_tests_properties(${name} PROPERTIES TIMEOUT 90)
  endforeach()
  qt_add_executable(ui_smoke tests/ui_smoke.cpp)
  target_link_libraries(ui_smoke PRIVATE flixify_desktop Qt6::Test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME package_policy COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s "${CMAKE_CURRENT_SOURCE_DIR}/scripts/tests" -v)
  set_tests_properties(package_policy PROPERTIES TIMEOUT 30)
endif()

install(TARGETS FlixifyWindows RUNTIME DESTINATION . BUNDLE DESTINATION .)
