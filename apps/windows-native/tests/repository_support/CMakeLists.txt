cmake_minimum_required(VERSION 3.28)
project(RepositoryBehaviorTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Concurrent Test)
add_library(repository_behavior STATIC ../../src/catalog.cpp ../../src/catalog.h ../../src/session_store.cpp ../../src/session_store.h ../../src/repository.cpp ../../src/repository.h ../../src/analytics.cpp ../../src/analytics.h ../../src/playback_analytics.cpp ../../src/playback_analytics.h)
target_include_directories(repository_behavior PUBLIC ../../src)
target_link_libraries(repository_behavior PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::Concurrent)
target_compile_definitions(repository_behavior PUBLIC FLIXIFY_APP_VERSION="1.0.2")
if(WIN32)
  target_link_libraries(repository_behavior PRIVATE crypt32)
endif()
foreach(name catalog repository)
  add_executable(${name}_test ../${name}_test.cpp)
  target_link_libraries(${name}_test PRIVATE repository_behavior Qt6::Test)
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
