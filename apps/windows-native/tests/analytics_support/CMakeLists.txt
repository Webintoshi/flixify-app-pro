cmake_minimum_required(VERSION 3.28)
project(FlixifyAnalyticsTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Network Test)
add_executable(analytics_test ../analytics_test.cpp)
target_include_directories(analytics_test PRIVATE ../../src)
target_compile_definitions(analytics_test PRIVATE FLIXIFY_APP_VERSION="1.0.2")
target_link_libraries(analytics_test PRIVATE Qt6::Core Qt6::Network Qt6::Test)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/analytics.cpp")
  target_sources(analytics_test PRIVATE ../../src/analytics.cpp ../../src/analytics.h)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/playback_analytics.cpp")
  target_sources(analytics_test PRIVATE ../../src/playback_analytics.cpp ../../src/playback_analytics.h)
endif()
